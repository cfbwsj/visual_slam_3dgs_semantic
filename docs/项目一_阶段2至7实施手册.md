### 2.2 创建 `src/vslam/common.py`

作用：共享坐标变换、投影、轨迹和 JSON/PLY 输出。库：NumPy、SciPy、标准库。

```python
import json
from pathlib import Path

import numpy as np
from scipy.spatial.transform import Rotation


def make_K(camera):
    return np.array([[camera['fx'], 0, camera['cx']],
                     [0, camera['fy'], camera['cy']],
                     [0, 0, 1]], dtype=np.float64)


def make_T(R, t):
    T = np.eye(4, dtype=np.float64)
    T[:3, :3] = R
    T[:3, 3] = np.asarray(t).reshape(3)
    return T


def transform(T, points):
    return np.asarray(points) @ T[:3, :3].T + T[:3, 3]


def project(K, points):
    z = points[:, 2]
    safe_z = np.where(np.abs(z) < 1e-9, 1e-9, z)
    uvh = points @ K.T
    return uvh[:, :2] / safe_z[:, None]


def backproject(uv, z, K):
    rays = np.column_stack([uv, np.ones(len(uv))]) @ np.linalg.inv(K).T
    return rays * np.asarray(z)[:, None]


def write_json(path, obj):
    path = Path(path)
    path.parent.mkdir(parents=True, exist_ok=True)
    path.write_text(json.dumps(obj, indent=2, ensure_ascii=False,
                               allow_nan=False), encoding='utf-8')


def write_tum(path, timestamps, poses_wc):
    rows = []
    for ts, T in zip(timestamps, poses_wc):
        q = Rotation.from_matrix(T[:3, :3]).as_quat()  # xyzw
        rows.append([ts, *T[:3, 3], *q])
    np.savetxt(path, np.asarray(rows), fmt='%.9f')


def read_tum(path):
    rows = np.loadtxt(path, comments='#', ndmin=2)
    if rows.shape[1] != 8 or not np.isfinite(rows).all():
        raise ValueError('轨迹必须是有限值的 TUM 八列格式')
    if np.any(np.diff(rows[:, 0]) <= 0):
        raise ValueError('轨迹时间戳必须严格递增')
    poses = np.array([make_T(Rotation.from_quat(row[4:8]).as_matrix(), row[1:4])
                      for row in rows])
    return rows[:, 0], poses


def nearest_pose(ts, times, poses, tolerance=0.02):
    # 这里用于按时间查姿态，不是 RGB-D 一对一关联。
    k = int(np.argmin(np.abs(times - ts)))
    return poses[k] if abs(times[k] - ts) <= tolerance else None


def write_ply(path, points, colors=None):
    points = np.asarray(points)
    colors = (np.full((len(points), 3), 180, dtype=np.uint8) if colors is None
              else np.clip(colors, 0, 255).astype(np.uint8))
    with Path(path).open('w', encoding='utf-8') as f:
        f.write('ply\nformat ascii 1.0\n')
        f.write(f'element vertex {len(points)}\n')
        f.write('property float x\nproperty float y\nproperty float z\n')
        f.write('property uchar red\nproperty uchar green\nproperty uchar blue\nend_header\n')
        for p, c in zip(points, colors):
            f.write(f'{p[0]} {p[1]} {p[2]} {int(c[0])} {int(c[1])} {int(c[2])}\n')
```

### 2.3 创建 `src/vslam/geometry/two_view.py`

作用：去畸变、本质矩阵、恢复位姿、三角化、正深度/视差/重投影过滤。库：OpenCV、NumPy。

```python
import cv2
import numpy as np

from vslam.common import make_T, project, transform


def estimate_pair(p0, p1, K, distortion, ransac_px=1.5,
                  reprojection_px=3.0, min_parallax_deg=0.5):
    if len(p0) < 8 or len(p0) != len(p1):
        raise ValueError('对应点不足或数量不一致')
    cv2.setRNGSeed(0)
    n0 = cv2.undistortPoints(p0.reshape(-1, 1, 2), K, distortion).reshape(-1, 2)
    n1 = cv2.undistortPoints(p1.reshape(-1, 1, 2), K, distortion).reshape(-1, 2)
    threshold = ransac_px / ((K[0, 0] + K[1, 1]) / 2)
    E, mask = cv2.findEssentialMat(n0, n1, np.eye(3), method=cv2.RANSAC,
                                  prob=0.999, threshold=threshold)
    if E is None or mask is None:
        raise RuntimeError('本质矩阵估计失败')
    # 某些输入会返回多个 3x3 候选，按 recoverPose 正深度内点数选取。
    if E.shape[1] != 3 or E.shape[0] % 3:
        raise RuntimeError(f'未识别的本质矩阵形状: {E.shape}')
    candidates = []
    for Ei in np.split(E, E.shape[0] // 3):
        count, R, t, pose_mask = cv2.recoverPose(
            Ei, n0, n1, np.eye(3), mask=mask.copy())
        candidates.append((count, R, t, pose_mask))
    count, R, t, pose_mask = max(candidates, key=lambda item: item[0])
    ids = np.flatnonzero(pose_mask.ravel() != 0)
    if len(ids) < 8:
        raise RuntimeError(f'恢复位姿后的有效点过少: {len(ids)}')
    P0 = np.column_stack([np.eye(3), np.zeros(3)])
    P1 = np.column_stack([R, t])
    Xh = cv2.triangulatePoints(P0, P1, n0[ids].T, n1[ids].T).T
    finite = np.abs(Xh[:, 3]) > 1e-10
    X = Xh[:, :3] / np.where(finite, Xh[:, 3], 1)[:, None]
    X1 = transform(make_T(R, t), X)
    # 与去畸变后的像素比较，避免混用畸变像素和针孔投影。
    uv0 = project(K, np.column_stack([n0[ids], np.ones(len(ids))]))
    uv1 = project(K, np.column_stack([n1[ids], np.ones(len(ids))]))
    e0 = np.linalg.norm(project(K, X) - uv0, axis=1)
    e1 = np.linalg.norm(project(K, X1) - uv1, axis=1)
    C1 = (-R.T @ t).ravel()
    ray0, ray1 = X, X - C1
    denominator = np.linalg.norm(ray0, axis=1) * np.linalg.norm(ray1, axis=1)
    cosine = np.sum(ray0 * ray1, axis=1) / np.maximum(denominator, 1e-12)
    angles = np.degrees(np.arccos(np.clip(cosine, -1, 1)))
    valid = (finite & np.isfinite(X).all(axis=1) & (X[:, 2] > 0)
             & (X1[:, 2] > 0) & (e0 < reprojection_px)
             & (e1 < reprojection_px) & (angles >= min_parallax_deg))
    if valid.sum() < 8:
        raise RuntimeError('过滤后点不足：检查视差、匹配、畸变和相机参数')
    report = {
        'matches': len(p0), 'ransac_inliers': int(np.count_nonzero(mask)),
        'recover_pose_inliers': int(count), 'valid_points': int(valid.sum()),
        'median_reprojection_px': float(np.median(np.maximum(e0, e1)[valid])),
        'median_parallax_deg': float(np.median(angles[valid])),
        'scale': 'arbitrary; baseline translation has unit norm',
    }
    result = dict(T_c1_c0=make_T(R, t), points_c0=X[valid],
                  uv0=uv0[valid], uv1=uv1[valid], K=K,
                  match_indices=ids[valid])
    return result, report
```

这里输出的 `uv0/uv1` 是去畸变像素，专门供后续 BA 使用。纯旋转和很小平移会使三角化退化；报失败是合理结果，不应不断放宽阈值强行得到点云。

### 2.4 创建 `src/vslam/cli/estimate_pair.py`

作用：读取阶段一输出，保存可复用的几何结果和误差图。库：argparse、NumPy、Matplotlib。

```python
import argparse
from pathlib import Path

import matplotlib
matplotlib.use('Agg')
import matplotlib.pyplot as plt
import numpy as np

from vslam.common import project, transform, write_json, write_ply
from vslam.geometry.two_view import estimate_pair


def main():
    p = argparse.ArgumentParser()
    p.add_argument('--input', required=True)
    p.add_argument('--out', required=True)
    args = p.parse_args()
    out = Path(args.out)
    out.mkdir(parents=True, exist_ok=True)
    with np.load(args.input) as d:
        result, report = estimate_pair(d['points0'], d['points1'], d['K'], d['distortion'])
        for key in ['timestamp0', 'timestamp1', 'index0', 'index1']:
            result[key] = d[key].copy()
    np.savez_compressed(out / 'geometry.npz', **result)
    write_ply(out / 'points_c0.ply', result['points_c0'])
    write_json(out / 'report.json', report)
    projected = project(result['K'], transform(result['T_c1_c0'], result['points_c0']))
    observed = result['uv1']
    fig, ax = plt.subplots()
    ax.scatter(observed[:, 0], observed[:, 1], s=4, label='observed undistorted')
    ax.scatter(projected[:, 0], projected[:, 1], s=4, label='reprojected')
    ax.invert_yaxis()
    ax.set_aspect('equal')
    ax.legend()
    fig.savefig(out / 'reprojection.png', dpi=150)
    plt.close(fig)
    print(report)


if __name__ == '__main__':
    main()
```

import numpy as np
from scipy.spatial.transform import Rotation

from vslam.common import make_T, project, transform, backproject
from vslam.geometry.two_view import estimate_pair


def test_projection_roundtrip():
    K = np.array([[500., 0, 320], [0, 500, 240], [0, 0, 1]])
    X = np.array([[0.2, -0.3, 2.], [-0.4, 0.5, 3.]])

    T = make_T(Rotation.from_rotvec([0.1, -0.05, 0.02]).as_matrix(), [0.4, 0.2, 0.1])

def test_two_view_known_scene():
    rng = np.random.default_rng(2)
    X = rng.uniform([-1., -0.8, 3.], [1., 0.8, 6.], size=(150, 3))
    K = np.array([[500., 0, 320], [0, 500, 240], [0, 0, 1]])

### 3.1 创建 `src/vslam/data/association.py`

作用：按绝对时间差从小到大选择一对一 RGB-D 对应；避免同一深度被重复使用。库：bisect、NumPy、argparse。适用于当前小数据集。

```python
import argparse
from bisect import bisect_left, bisect_right
from pathlib import Path

import numpy as np

from vslam.data.tum import read_index


def associate(rgb_rows, depth_rows, max_difference=0.02):
    if max_difference <= 0:
        raise ValueError('时间阈值必须为正')
    for rows in [rgb_rows, depth_rows]:
        ts = np.array([x[0] for x in rows])
        if not len(ts) or np.any(np.diff(ts) <= 0):
            raise ValueError('索引为空或时间戳不严格递增')
    times_d = [x[0] for x in depth_rows]
    candidates = []
    for i, (t, _) in enumerate(rgb_rows):
        lo = bisect_left(times_d, t - max_difference)
        hi = bisect_right(times_d, t + max_difference)
        candidates.extend((abs(t - times_d[j]), i, j) for j in range(lo, hi))
    used_rgb, used_depth, selected = set(), set(), []
    for delta, i, j in sorted(candidates):
        if i not in used_rgb and j not in used_depth:
            selected.append((*rgb_rows[i], *depth_rows[j]))
            used_rgb.add(i)
            used_depth.add(j)
    return sorted(selected, key=lambda row: row[0])


def read_associations(path):
    rows = []
    for line in Path(path).read_text().splitlines():
        if not line.strip() or line.lstrip().startswith('#'):
            continue
        tr, rgb, td, depth = line.split()
        rows.append((float(tr), rgb, float(td), depth))
    if not rows:
        raise ValueError('没有 RGB-D 关联')
    return rows


def main():
    p = argparse.ArgumentParser()
    p.add_argument('--root', required=True)
    p.add_argument('--out', required=True)
    p.add_argument('--max-difference', type=float, default=0.02)
    args = p.parse_args()
    root = Path(args.root)
    rows = associate(read_index(root / 'rgb.txt'), read_index(root / 'depth.txt'),
                     args.max_difference)
    if not rows:
        raise RuntimeError('没有满足时间差的对应帧')
    out = Path(args.out)
    out.parent.mkdir(parents=True, exist_ok=True)
    with out.open('w') as f:
        for tr, rgb, td, depth in rows:
            f.write(f'{tr:.6f} {rgb} {td:.6f} {depth}\n')
    gaps = np.array([abs(row[0] - row[2]) for row in rows])
    print({'pairs': len(rows), 'max_gap_s': float(gaps.max()),
           'median_gap_s': float(np.median(gaps))})


if __name__ == '__main__':
    main()
```

  min_inliers: 15
  min_inlier_ratio: 0.35
  reprojection_px: 3.0
  max_step_m: 0.5
  max_rotation_deg: 30.0

### 3.3 创建 `src/vslam/cli/run_vo.py`

作用：完整短序列 RGB-D VO；失败即停止当前连续轨迹，并保留失败日志。库：OpenCV、NumPy、PyYAML、标准库。

```python
import argparse
import csv
import time
from pathlib import Path

import cv2
import numpy as np
import yaml

from vslam.common import backproject, make_K, make_T, write_json, write_tum
from vslam.config import load_config
from vslam.data.association import read_associations
from vslam.vo.features import extract_orb, match_ratio, matched_points


def load_rgbd(root, row, shape):
    rgb = cv2.imread(str(root / row[1]), cv2.IMREAD_COLOR)
    depth = cv2.imread(str(root / row[3]), cv2.IMREAD_UNCHANGED)
    if rgb is None or depth is None:
        raise FileNotFoundError(f'读取失败: {row}')
    if rgb.shape[:2] != shape or depth.shape != shape or depth.dtype != np.uint16:
        raise ValueError('图像尺寸或深度格式不正确')
    return rgb, depth


def main():
    p = argparse.ArgumentParser()
    p.add_argument('--config', required=True)
    p.add_argument('--associations', required=True)
    p.add_argument('--max-frames', type=int, default=100)
    p.add_argument('--out', required=True)
    args = p.parse_args()
    cfg = load_config(args.config)
    root, out = Path(cfg['dataset']['root']), Path(args.out)
    out.mkdir(parents=True, exist_ok=True)
    rows = read_associations(args.associations)
    if args.max_frames > 0:
        rows = rows[:args.max_frames]
    if len(rows) < 2:
        raise ValueError('至少需要两组 RGB-D')
    K, v = make_K(cfg['camera']), cfg['vo']
    if np.any(cfg['camera']['distortion']):
        raise ValueError('本 RGB-D 基线要求预配准深度 + 零畸变模型')
    shape = (cfg['camera']['height'], cfg['camera']['width'])
    cv2.setRNGSeed(0)
    image0, depth0 = load_rgbd(root, rows[0], shape)
    kp0, des0 = extract_orb(image0, cfg['orb']['nfeatures'])
    poses, stamps, logs = [np.eye(4)], [rows[0][0]], []
    stopped = False
    for row in rows[1:]:
        start = time.perf_counter()
        log = dict(timestamp=row[0], matches=0, depth_points=0, inliers=0,
                   median_error_px=None, elapsed_ms=0, status='ok', reason='')
        try:
            image1, depth1 = load_rgbd(root, row, shape)
            kp1, des1 = extract_orb(image1, cfg['orb']['nfeatures'])
            matches = match_ratio(des0, des1, cfg['matching']['ratio'])
            uv0, uv1 = matched_points(kp0, kp1, matches)
            log['matches'] = len(matches)
            ij = np.rint(uv0).astype(int)
            inside = ((ij[:, 0] >= 0) & (ij[:, 0] < shape[1])
                      & (ij[:, 1] >= 0) & (ij[:, 1] < shape[0]))
            ij, uv0, uv1 = ij[inside], uv0[inside], uv1[inside]
            z = depth0[ij[:, 1], ij[:, 0]].astype(np.float64) / cfg['camera']['depth_scale']
            valid = (z >= v['min_depth']) & (z <= v['max_depth'])
            X = backproject(uv0[valid], z[valid], K)
            target = uv1[valid]
            log['depth_points'] = len(X)
            if len(X) < v['min_correspondences']:
                raise RuntimeError('有效深度对应不足')
            ok, rvec, tvec, ids = cv2.solvePnPRansac(
                X, target, K, np.zeros(5), iterationsCount=200,
                reprojectionError=v['reprojection_px'], confidence=0.999,
                flags=cv2.SOLVEPNP_EPNP)
            if not ok or ids is None:
                raise RuntimeError('PnP RANSAC 失败')
            ids = ids.ravel()
            log['inliers'] = len(ids)
            if len(ids) < v['min_inliers'] or len(ids) / len(X) < v['min_inlier_ratio']:
                raise RuntimeError('PnP 内点数或比例不足')
            rvec, tvec = cv2.solvePnPRefineLM(X[ids], target[ids], K, np.zeros(5), rvec, tvec)
            pixels, _ = cv2.projectPoints(X[ids], rvec, tvec, K, np.zeros(5))
            error = np.linalg.norm(pixels.reshape(-1, 2) - target[ids], axis=1)
            log['median_error_px'] = float(np.median(error))
            if not np.isfinite(error).all() or np.median(error) > v['reprojection_px']:
                raise RuntimeError('优化后重投影误差异常')
            if np.linalg.norm(tvec) > v['max_step_m']:
                raise RuntimeError('单步平移异常')
            if np.degrees(np.linalg.norm(rvec)) > v['max_rotation_deg']:
                raise RuntimeError('单步旋转异常')
            T_c1_c0 = make_T(cv2.Rodrigues(rvec)[0], tvec)
            poses.append(poses[-1] @ np.linalg.inv(T_c1_c0))
            stamps.append(row[0])
            kp0, des0, depth0 = kp1, des1, depth1
        except (RuntimeError, ValueError, FileNotFoundError, cv2.error) as exc:
            log['status'], log['reason'], stopped = 'lost', str(exc), True
        log['elapsed_ms'] = (time.perf_counter() - start) * 1000
        logs.append(log)
        if stopped:
            break
    with (out / 'tracking.csv').open('w', newline='') as f:
        writer = csv.DictWriter(f, fieldnames=list(logs[0]))
        writer.writeheader()
        writer.writerows(logs)
    write_tum(out / 'trajectory_tum.txt', stamps, poses)
    np.savez_compressed(out / 'trajectory.npz', timestamps=stamps, poses_wc=poses)
    (out / 'config_used.yaml').write_text(yaml.safe_dump(cfg), encoding='utf-8')
    report = dict(requested_frames=len(rows), successful_poses=len(poses),
                  coverage=len(poses) / len(rows), stopped_on_failure=stopped,
                  elapsed_ms_mean=float(np.mean([x['elapsed_ms'] for x in logs])))
    write_json(out / 'report.json', report)
    print(report)
    if stopped:
        raise SystemExit('跟踪停止；检查 tracking.csv 最后一行，未伪造后续位姿')


if __name__ == '__main__':
    main()
```

## 5. 阶段 4：Bundle Adjustment

目标：先在独立合成场景中验证优化，再读取阶段 2A 的几何输出。第一版使用 SciPy 稀疏有限差分和非线性最小二乘，后续再推导解析雅可比或移植 Ceres。

### 5.1 创建 `src/vslam/optimization/ba.py`

库：NumPy、SciPy、OpenCV。输入观测为无畸变像素。

为消除 gauge freedom，固定第一相机位姿和第一个点在该坐标系中的 z。合成实验使用已知 z 锚定米制尺度；真实两帧使用初始 z 锚定已有任意尺度，不能因此声称恢复了米制尺度。

```python
import argparse
from pathlib import Path

import cv2
import numpy as np
from scipy.optimize import least_squares
from scipy.sparse import lil_matrix
from scipy.spatial.transform import Rotation

from vslam.common import make_T, project, transform, write_json, write_ply


def synthetic(seed=0, outlier_fraction=0.0):
    rng = np.random.default_rng(seed)
    K = np.array([[500., 0, 320], [0, 500, 240], [0, 0, 1]])
    X = rng.uniform([-1., -0.8, 3.], [1., 0.8, 5.], size=(60, 3))
    cameras = np.zeros((4, 6))  # rvec, tvec; 世界到相机
    cameras[:, 3] = -np.arange(4) * 0.25
    ci = np.repeat(np.arange(4), len(X))
    pi = np.tile(np.arange(len(X)), 4)
    clean = np.concatenate([project(K, X + c[3:]) for c in cameras])
    uv = clean + rng.normal(0, 0.5, clean.shape)
    bad = rng.choice(len(uv), int(len(uv) * outlier_fraction), replace=False)
    uv[bad] += rng.normal(0, 25, (len(bad), 2))
    initial_cameras = cameras.copy()
    initial_cameras[1:, :3] += rng.normal(0, 0.01, (3, 3))
    initial_cameras[1:, 3:] += rng.normal(0, 0.025, (3, 3))
    initial_X = X + rng.normal(0, 0.03, X.shape)
    initial_X[0, 2] = X[0, 2]
    return K, initial_cameras, initial_X, ci, pi, uv, clean


def from_pair(path):
    with np.load(path) as d:
        X = d['points_c0'].copy()
        # 小型实验限制规模，以便在 CPU 上调试。
        ids = np.linspace(0, len(X) - 1, min(120, len(X)), dtype=int)
        X = X[ids]
        T = d['T_c1_c0']
        cameras = np.zeros((2, 6))
        cameras[1, :3] = cv2.Rodrigues(T[:3, :3])[0].ravel()
        cameras[1, 3:] = T[:3, 3]
        uv = np.concatenate([d['uv0'][ids], d['uv1'][ids]])
        return (d['K'].copy(), cameras, X, np.repeat([0, 1], len(X)),
                np.tile(np.arange(len(X)), 2), uv, None)


def optimize(K, cameras0, points0, ci, pi, uv, loss):
    ncam, npoint = len(cameras0), len(points0)
    camera_variables = 6 * (ncam - 1)
    free_point_coordinates = np.delete(np.arange(points0.size), 2)
    # 固定第一点 z；其余点坐标都是未知量。
    x0 = np.r_[cameras0[1:].ravel(), points0.ravel()[free_point_coordinates]]

    def unpack(x):
        cameras = cameras0.copy()
        cameras[1:] = x[:camera_variables].reshape(-1, 6)
        points = points0.ravel().copy()
        points[free_point_coordinates] = x[camera_variables:]
        return cameras, points.reshape(-1, 3)

    def prediction(x):
        cameras, points = unpack(x)
        R = Rotation.from_rotvec(cameras[:, :3]).as_matrix()
        Xc = np.einsum('nij,nj->ni', R[ci], points[pi]) + cameras[ci, 3:]
        return project(K, Xc), Xc[:, 2]

    def residual(x):
        pixels, _ = prediction(x)
        return (pixels - uv).ravel()

    sparsity = lil_matrix((2 * len(uv), len(x0)), dtype=int)
    point_lookup = {int(old): camera_variables + j
                    for j, old in enumerate(free_point_coordinates)}
    for obs, (c, pt) in enumerate(zip(ci, pi)):
        for row in (2 * obs, 2 * obs + 1):
            if c > 0:
                sparsity[row, 6 * (c - 1):6 * c] = 1
            for old in range(3 * pt, 3 * pt + 3):
                if old in point_lookup:
                    sparsity[row, point_lookup[old]] = 1
    result = least_squares(residual, x0, jac_sparsity=sparsity.tocsr(),
                           method='trf', x_scale='jac', loss=loss,
                           f_scale=1.0, max_nfev=200)
    pred0, _ = prediction(x0)
    pred1, z = prediction(result.x)
    if not np.isfinite(pred1).all() or np.any(z <= 0):
        raise RuntimeError('优化后有非有限值或负深度，不能接收此解')
    cameras, points = unpack(result.x)
    report = dict(success=bool(result.success), message=str(result.message),
                  evaluations=int(result.nfev), loss=loss,
                  before_rmse_px=float(np.sqrt(np.mean((pred0 - uv) ** 2))),
                  after_rmse_px=float(np.sqrt(np.mean((pred1 - uv) ** 2))),
                  before_median_px=float(np.median(np.linalg.norm(pred0 - uv, axis=1))),
                  after_median_px=float(np.median(np.linalg.norm(pred1 - uv, axis=1))))
    return cameras, points, pred0, pred1, report


def main():
    p = argparse.ArgumentParser()
    p.add_argument('--pair', help='不提供时使用合成数据')
    p.add_argument('--loss', choices=['linear', 'huber'], default='huber')
    p.add_argument('--outliers', type=float, default=0.0)
    p.add_argument('--out', required=True)
    args = p.parse_args()
    if not 0 <= args.outliers < 1:
        raise ValueError('外点比例必须在 [0, 1)')
    data = from_pair(args.pair) if args.pair else synthetic(outlier_fraction=args.outliers)
    K, cams, points, ci, pi, uv, clean = data
    c1, X1, pred0, pred1, report = optimize(K, cams, points, ci, pi, uv, args.loss)
    report['scale'] = 'arbitrary' if args.pair else 'synthetic metric anchor'
    if clean is not None:
        report['clean_rmse_before'] = float(np.sqrt(np.mean((pred0 - clean) ** 2)))
        report['clean_rmse_after'] = float(np.sqrt(np.mean((pred1 - clean) ** 2)))
    out = Path(args.out)
    out.mkdir(parents=True, exist_ok=True)
    np.savez_compressed(out / 'ba.npz', K=K, cameras_before=cams, cameras_after=c1,
                        points_before=points, points_after=X1,
                        camera_indices=ci, point_indices=pi, observations=uv)
    write_ply(out / 'before.ply', points)
    write_ply(out / 'after.ply', X1)
    write_json(out / 'report.json', report)
    print(report)
    if not report['success']:
        raise SystemExit('已保存候选结果，但求解未收敛；查看 report 后再决定是否增加迭代')


if __name__ == '__main__':
    main()
```

### 6.2 创建 `src/vslam/reconstruction/select_frames.py`

作用：从 TUM 索引按间隔选择图像，保留时间戳与原始文件映射。库：OpenCV、csv、shutil。清晰度阈值只作粗筛，最终人工检查运动模糊和覆盖。

```python
import argparse
import csv
import shutil
from pathlib import Path
import cv2
from vslam.data.tum import read_index


def main():
    p = argparse.ArgumentParser()
    p.add_argument('--root', required=True)
    p.add_argument('--out', required=True)
    p.add_argument('--stride', type=int, default=8)
    p.add_argument('--max-frames', type=int, default=100)
    p.add_argument('--min-sharpness', type=float, default=0.0)
    args = p.parse_args()
    if args.stride < 1 or args.max_frames < 1:
        raise ValueError('stride 和 max-frames 必须为正')
    root, out = Path(args.root), Path(args.out)
    image_dir = out / 'input'
    if image_dir.exists() and any(image_dir.iterdir()):
        raise FileExistsError('请使用新的场景目录，避免混入旧图片')
    image_dir.mkdir(parents=True, exist_ok=True)
    records = []
    for i, (ts, rel) in enumerate(read_index(root / 'rgb.txt')):
        if i % args.stride:
            continue
        image = cv2.imread(str(root / rel), cv2.IMREAD_GRAYSCALE)
        if image is None:
            raise FileNotFoundError(root / rel)
        sharpness = float(cv2.Laplacian(image, cv2.CV_64F).var())
        if sharpness < args.min_sharpness:
            continue
        name = f'{i:06d}{Path(rel).suffix}'
        shutil.copy2(root / rel, image_dir / name)
        records.append(dict(name=name, timestamp=ts, original=rel, sharpness=sharpness))
        if len(records) >= args.max_frames:
            break
    if len(records) < 10:
        raise RuntimeError('清晰图片不足 10 张，检查筛选条件')
    with (out / 'frames.csv').open('w', newline='') as f:
        writer = csv.DictWriter(f, fieldnames=list(records[0]))
        writer.writeheader()
        writer.writerows(records)
    print(f'已选择 {len(records)} 张图像')


if __name__ == '__main__':
    main()
```

### 7.2 创建 `src/vslam/semantic/build_map.py`

作用：分割区域、编码区域特征、使用有效深度和姿态投到世界坐标、按体素融合。保存可重查的区域输出，且限定体素数量，避免特征数组无限增长。库：SAM、CLIP、PyTorch、NumPy、OpenCV、Pillow。

```python
import argparse
import gc
from pathlib import Path

import clip
import cv2
import numpy as np
import torch
from PIL import Image
from segment_anything import SamAutomaticMaskGenerator, sam_model_registry

from vslam.common import (backproject, make_K, nearest_pose, read_tum,
                           transform, write_json, write_ply)
from vslam.config import load_config
from vslam.data.association import read_associations
from vslam.cli.run_vo import load_rgbd


def release_cuda():
    gc.collect()
    if torch.cuda.is_available():
        torch.cuda.empty_cache()


def main():
    p = argparse.ArgumentParser()
    p.add_argument('--config', required=True)
    p.add_argument('--associations', required=True)
    p.add_argument('--trajectory', required=True)
    p.add_argument('--checkpoint', required=True)
    p.add_argument('--clip-cache', default='checkpoints/clip')
    p.add_argument('--device', choices=['cpu', 'cuda'], default='cpu')
    p.add_argument('--max-frames', type=int, default=1)
    p.add_argument('--frame-stride', type=int, default=15)
    p.add_argument('--pixel-stride', type=int, default=8)
    p.add_argument('--voxel-size', type=float, default=0.03)
    p.add_argument('--max-voxels', type=int, default=50000)
    p.add_argument('--out', required=True)
    args = p.parse_args()
    if min(args.max_frames, args.frame_stride, args.pixel_stride, args.max_voxels) < 1 or args.voxel_size <= 0:
        raise ValueError('采样参数必须为正')
    if args.device == 'cuda' and not torch.cuda.is_available():
        raise RuntimeError('CUDA 不可用，先使用 --device cpu')
    cfg = load_config(args.config)
    if np.any(cfg['camera']['distortion']):
        raise ValueError('当前融合只接受预配准深度的零畸变模型')
    root, out = Path(cfg['dataset']['root']), Path(args.out)
    out.mkdir(parents=True, exist_ok=True)
    (out / 'frames').mkdir(exist_ok=True)
    K = make_K(cfg['camera'])
    shape = (cfg['camera']['height'], cfg['camera']['width'])
    times, poses = read_tum(args.trajectory)
    rows = read_associations(args.associations)[::args.frame_stride]
    sam = sam_model_registry['vit_b'](checkpoint=args.checkpoint).eval()
    generator = SamAutomaticMaskGenerator(sam, points_per_side=16,
                                         points_per_batch=16, min_mask_region_area=0)
    model, preprocess = clip.load('ViT-B/32', device='cpu', download_root=args.clip_cache)
    model.eval()
    table, used_frames, skipped_pose, frame_reports = {}, 0, 0, []
    for frame_id, row in enumerate(rows):
        T = nearest_pose(row[0], times, poses, tolerance=0.02)
        if T is None:
            skipped_pose += 1
            continue
        bgr, depth = load_rgbd(root, row, shape)
        rgb = cv2.cvtColor(bgr, cv2.COLOR_BGR2RGB)
        sam.to(args.device)
        with torch.inference_mode():
            masks = generator.generate(rgb)
        sam.cpu()
        release_cuda()
        # 让较小的区域优先占有像素；这一启发式要在报告中说明。
        masks = sorted([m for m in masks if 200 <= m['area'] <= 0.8 * rgb.shape[0] * rgb.shape[1]],
                       key=lambda m: m['area'])[:50]
        labels = np.full(depth.shape, -1, dtype=np.int32)
        features, qualities = [], []
        model.to(args.device)
        for region_id, mask in enumerate(masks):
            segmentation = mask['segmentation']
            yy, xx = np.where(segmentation)
            x0, x1, y0, y1 = xx.min(), xx.max() + 1, yy.min(), yy.max() + 1
            crop = rgb[y0:y1, x0:x1].copy()
            crop[~segmentation[y0:y1, x0:x1]] = 127
            tensor = preprocess(Image.fromarray(crop)).unsqueeze(0).to(args.device)
            with torch.inference_mode():
                feature = model.encode_image(tensor).float()
                feature = feature / feature.norm(dim=-1, keepdim=True).clamp_min(1e-8)
            features.append(feature.cpu().numpy()[0])
            qualities.append(float(np.clip(mask['predicted_iou'], 0.05, 1.0)))
            labels[segmentation & (labels < 0)] = region_id
            del tensor, feature
        model.cpu()
        release_cuda()
        if not features:
            frame_reports.append(dict(timestamp=row[0], regions=0))
            continue
        features = np.asarray(features, dtype=np.float32)
        yy, xx = np.mgrid[0:shape[0]:args.pixel_stride, 0:shape[1]:args.pixel_stride]
        xx, yy = xx.ravel(), yy.ravel()
        z = depth[yy, xx].astype(np.float64) / cfg['camera']['depth_scale']
        region_ids = labels[yy, xx]
        valid = ((z >= cfg['vo']['min_depth']) & (z <= cfg['vo']['max_depth']) & (region_ids >= 0))
        xx, yy, z, region_ids = xx[valid], yy[valid], z[valid], region_ids[valid]
        world = transform(T, backproject(np.column_stack([xx, yy]), z, K))
        keys = np.floor(world / args.voxel_size).astype(np.int64)
        # 同一帧同一体素只贡献一次，避免高像素密度造成虚假的多视角权重。
        seen = set()
        for key_array, point, color, rid in zip(keys, world, rgb[yy, xx], region_ids):
            key = tuple(key_array.tolist())
            if key in seen:
                continue
            seen.add(key)
            if key not in table:
                if len(table) >= args.max_voxels:
                    raise RuntimeError('体素数量超过限制：减少帧数或增大体素尺寸后重跑')
                table[key] = [np.zeros(3), np.zeros(3), np.zeros(features.shape[1], dtype=np.float32), 0., 0]
            accumulator = table[key]
            weight = qualities[rid]
            accumulator[0] += weight * point
            accumulator[1] += weight * color
            accumulator[2] += weight * features[rid]
            accumulator[3] += weight
            accumulator[4] += 1
        np.savez_compressed(out / 'frames' / f'{frame_id:06d}.npz', timestamp=row[0],
                            labels=labels, region_features=features, pose_wc=T)
        preview = rgb.copy()
        rng = np.random.default_rng(0)
        palette = rng.integers(0, 255, (len(features), 3), dtype=np.uint8)
        mask_valid = labels >= 0
        preview[mask_valid] = (0.5 * rgb[mask_valid] + 0.5 * palette[labels[mask_valid]]).astype(np.uint8)
        cv2.imwrite(str(out / 'frames' / f'{frame_id:06d}.jpg'), cv2.cvtColor(preview, cv2.COLOR_RGB2BGR))
        frame_reports.append(dict(timestamp=row[0], regions=len(features), sampled_points=len(world)))
        used_frames += 1
        print(f'frames={used_frames}, voxels={len(table)}')
        if used_frames >= args.max_frames:
            break
    if not table:
        raise RuntimeError('没有有效语义点：检查姿态时间、深度和分割')
    values = list(table.values())
    points = np.array([a[0] / a[3] for a in values])
    colors = np.array([a[1] / a[3] for a in values]).astype(np.uint8)
    features = np.array([a[2] / a[3] for a in values], dtype=np.float32)
    features /= np.maximum(np.linalg.norm(features, axis=1, keepdims=True), 1e-8)
    observations = np.array([a[4] for a in values], dtype=np.int32)
    np.savez_compressed(out / 'semantic_map.npz', points=points, colors=colors,
                        features=features, observations=observations,
                        clip_model='ViT-B/32', trajectory=str(Path(args.trajectory).resolve()),
                        voxel_size=args.voxel_size, units='meters')
    write_ply(out / 'semantic_map.ply', points, colors)
    write_json(out / 'report.json', dict(frames=used_frames, skipped_missing_pose=skipped_pose,
                                        voxels=len(points), frame_reports=frame_reports,
                                        arguments=vars(args)))


if __name__ == '__main__':
    main()
```

这版是“区域裁剪的 CLIP 特征传播到可见深度点”。不是像素级特征，也不是实例跟踪。没有 mask 的点不强制赋值；混合边界、遮挡和区域裁剪偏差会造成误差，应保留示例分析。

### 7.3 创建 `src/vslam/semantic/query.py`

作用：文字查询、保存相似度、导出高亮点云。使用候选类别相对分数与绝对余弦阈值，不把 softmax 当成校准概率。

```python
import argparse
from pathlib import Path

import clip
import numpy as np
import torch

from vslam.common import write_json, write_ply


def main():
    p = argparse.ArgumentParser()
    p.add_argument('--map', required=True)
    p.add_argument('--text', default='chair')
    p.add_argument('--alternatives', nargs='+', default=['table', 'wall', 'floor', 'monitor'])
    p.add_argument('--threshold', type=float, default=0.20)
    p.add_argument('--min-observations', type=int, default=1)
    p.add_argument('--clip-cache', default='checkpoints/clip')
    p.add_argument('--out', required=True)
    args = p.parse_args()
    with np.load(args.map) as d:
        points, features = d['points'].copy(), d['features'].copy()
        counts = d['observations'].copy()
        model_name = str(d['clip_model'])
    model, _ = clip.load(model_name, device='cpu', download_root=args.clip_cache)
    names = [args.text] + [x for x in args.alternatives if x != args.text]
    with torch.inference_mode():
        text = model.encode_text(clip.tokenize([f'a photo of a {x}' for x in names])).float()
        text /= text.norm(dim=-1, keepdim=True).clamp_min(1e-8)
    scores = features @ text.numpy().T
    selected = ((scores.argmax(axis=1) == 0) & (scores[:, 0] >= args.threshold)
                & (counts >= args.min_observations))
    colors = np.full((len(points), 3), 100, dtype=np.uint8)
    colors[selected] = [255, 60, 30]
    out = Path(args.out)
    out.mkdir(parents=True, exist_ok=True)
    write_ply(out / 'query.ply', points, colors)
    np.savez_compressed(out / 'scores.npz', scores=scores, selected=selected, names=np.array(names))
    write_json(out / 'report.json', dict(query=args.text, selected_points=int(selected.sum()),
                                        total_points=len(points), threshold=args.threshold,
                                        candidates=names))


if __name__ == '__main__':
    main()
```
