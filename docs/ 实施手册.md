### 2.2 创建 `src/vslam/common.py`

作用：集中管理相机内参、坐标变换、投影/反投影、TUM 轨迹和点云文件，避免各模块重复实现。主要库：NumPy 负责矩阵运算；SciPy `Rotation` 负责旋转矩阵与四元数转换；`json`、`Path` 负责文件输出。

关键函数：`transform()` 执行刚体变换，`project()` 将三维点投影到图像，`backproject()` 根据像素和深度恢复三维点；`write_tum()` / `read_tum()` 负责 TUM 轨迹读写。

```python
import json
from pathlib import Path

import numpy as np
from scipy.spatial.transform import Rotation


def make_K(camera):
    # fx、fy 是焦距，cx、cy 是主点坐标。
    return np.array([[camera["fx"], 0, camera["cx"]],
                     [0, camera["fy"], camera["cy"]],
                     [0, 0, 1]], dtype=np.float64)


def make_T(R, t):
    # 将 R、t 组合成 4×4 齐次变换矩阵。
    T = np.eye(4)
    T[:3, :3], T[:3, 3] = R, np.asarray(t).reshape(3)
    return T


def transform(T, points):
    # 对 Nx3 点集执行 p' = R p + t。
    return np.asarray(points) @ T[:3, :3].T + T[:3, 3]


def project(K, points):
    # 针孔模型：相机三维坐标投影到像素平面。
    uvh = points @ K.T
    return uvh[:, :2] / uvh[:, 2:3]


def backproject(uv, z, K):
    # 像素坐标乘 K^-1 得到视线方向，再乘深度得到三维点。
    rays = np.column_stack([uv, np.ones(len(uv))]) @ np.linalg.inv(K).T
    return rays * np.asarray(z)[:, None]


def write_json(path, obj):
    path = Path(path)
    path.parent.mkdir(parents=True, exist_ok=True)
    path.write_text(json.dumps(obj, indent=2, ensure_ascii=False), encoding="utf-8")


def write_tum(path, timestamps, poses_wc):
    rows = []
    for ts, T in zip(timestamps, poses_wc):
        # SciPy 输出四元数顺序为 xyzw，正好符合 TUM 格式。
        q = Rotation.from_matrix(T[:3, :3]).as_quat()
        rows.append([ts, *T[:3, 3], *q])
    path = Path(path)
    path.parent.mkdir(parents=True, exist_ok=True)
    np.savetxt(path, rows, fmt="%.9f")


def read_tum(path):
    # TUM 每行：timestamp tx ty tz qx qy qz qw。
    rows = np.loadtxt(path, comments="#", ndmin=2)
    poses = np.array([
        make_T(Rotation.from_quat(row[4:8]).as_matrix(), row[1:4])
        for row in rows
    ])
    return rows[:, 0], poses


def nearest_pose(ts, times, poses, tolerance=0.02):
    # 找最近时间戳；时间差超过 tolerance 秒就不关联。
    index = np.argmin(np.abs(times - ts))
    return poses[index] if abs(times[index] - ts) <= tolerance else None


def write_ply(path, points, colors=None):
    # 保存 ASCII PLY；未提供颜色时统一使用灰色。
    points = np.asarray(points)
    if colors is None:
        colors = np.full((len(points), 3), 180, dtype=np.uint8)
    else:
        colors = np.clip(colors, 0, 255).astype(np.uint8)

    path = Path(path)
    path.parent.mkdir(parents=True, exist_ok=True)
    with path.open("w", encoding="utf-8") as f:
        f.write("ply\nformat ascii 1.0\n")
        f.write(f"element vertex {len(points)}\n")
        f.write("property float x\nproperty float y\nproperty float z\n")
        f.write("property uchar red\nproperty uchar green\nproperty uchar blue\nend_header\n")
        for p, c in zip(points, colors):
            f.write(f"{p[0]} {p[1]} {p[2]} {int(c[0])} {int(c[1])} {int(c[2])}\n")
```

### 2.3 创建 `src/vslam/geometry/two_view.py`

作用：从两帧匹配点估计相对旋转和平移，再三角化出第一帧坐标系下的三维点。主要库：OpenCV 提供几何估计，NumPy 负责矩阵和筛选运算。

关键 API：`undistortPoints()` 去畸变；`findEssentialMat()` 用 RANSAC 估计本质矩阵；`recoverPose()` 恢复相对位姿并利用正深度筛选；`triangulatePoints()` 三角化。平移方向可恢复，但长度归一化，因此尺度仍然任意。

```python
import cv2
import numpy as np

from vslam.common import make_T, project, transform


def estimate_pair(p0, p1, K, distortion, ransac_px=1.5,
                  reprojection_px=3.0, min_parallax_deg=0.5):
    if len(p0) < 8 or len(p0) != len(p1):
        raise ValueError("对应点不足或数量不一致")

    # undistortPoints 去除镜头畸变，输出归一化相机坐标。
    n0 = cv2.undistortPoints(p0.reshape(-1, 1, 2), K, distortion).reshape(-1, 2)
    n1 = cv2.undistortPoints(p1.reshape(-1, 1, 2), K, distortion).reshape(-1, 2)

    # findEssentialMat 用 RANSAC 估计本质矩阵 E，并标记几何内点。
    threshold = ransac_px / ((K[0, 0] + K[1, 1]) / 2)
    E, mask = cv2.findEssentialMat(
        n0, n1, np.eye(3), method=cv2.RANSAC, prob=0.999, threshold=threshold
    )
    if E is None or mask is None:
        raise RuntimeError("本质矩阵估计失败")

    # E 可能包含多个 3×3 候选；recoverPose 用正深度约束选出可行位姿。
    solutions = [
        cv2.recoverPose(Ei, n0, n1, np.eye(3), mask=mask.copy())
        for Ei in np.split(E, E.shape[0] // 3)
    ]
    count, R, t, pose_mask = max(solutions, key=lambda item: item[0])
    ids = np.flatnonzero(pose_mask.ravel())

    # triangulatePoints 输入两个投影矩阵和对应归一化坐标，输出齐次三维点。
    P0 = np.column_stack([np.eye(3), np.zeros(3)])
    P1 = np.column_stack([R, t])
    Xh = cv2.triangulatePoints(P0, P1, n0[ids].T, n1[ids].T).T
    finite = np.abs(Xh[:, 3]) > 1e-10
    X = Xh[:, :3] / Xh[:, 3:4]
    X1 = transform(make_T(R, t), X)

    # 将归一化坐标乘 K 转回像素坐标，计算两帧重投影误差。
    uv0 = project(K, np.column_stack([n0[ids], np.ones(len(ids))]))
    uv1 = project(K, np.column_stack([n1[ids], np.ones(len(ids))]))
    e0 = np.linalg.norm(project(K, X) - uv0, axis=1)
    e1 = np.linalg.norm(project(K, X1) - uv1, axis=1)

    # 两条视线夹角越小，三角化深度越不稳定。
    C1 = (-R.T @ t).ravel()
    ray0, ray1 = X, X - C1
    denominator = np.linalg.norm(ray0, axis=1) * np.linalg.norm(ray1, axis=1)
    cosine = np.sum(ray0 * ray1, axis=1) / denominator
    angles = np.degrees(np.arccos(np.clip(cosine, -1, 1)))

    # 保留正深度、有限坐标、低重投影误差和足够视差的点。
    valid = (finite & np.isfinite(X).all(axis=1) & (X[:, 2] > 0)
             & (X1[:, 2] > 0) & (e0 < reprojection_px)
             & (e1 < reprojection_px) & (angles >= min_parallax_deg))
    if valid.sum() < 8:
        raise RuntimeError("过滤后有效三维点不足")

    report = {
        "matches": len(p0),
        "ransac_inliers": int(np.count_nonzero(mask)),
        "recover_pose_inliers": int(count),
        "valid_points": int(valid.sum()),
        "median_reprojection_px": float(np.median(np.maximum(e0, e1)[valid])),
        "median_parallax_deg": float(np.median(angles[valid])),
        "scale": "arbitrary; baseline translation has unit norm"
    }
    result = dict(
        T_c1_c0=make_T(R, t), points_c0=X[valid],
        uv0=uv0[valid], uv1=uv1[valid], K=K, match_indices=ids[valid]
    )
    return result, report

```

这里输出的 `uv0/uv1` 是去畸变像素，专门供后续 BA 使用。纯旋转和很小平移会使三角化退化；报失败是合理结果，不应不断放宽阈值强行得到点云。

### 2.4 创建 `src/vslam/cli/estimate_pair.py`

作用：把两帧几何算法接到命令行，保存 `geometry.npz`、PLY 点云、JSON 报告和重投影图。`np.load()` 读取阶段一结果，`np.savez_compressed()` 保存后续 BA 可复用的数组；Matplotlib 的 `scatter()` 用于对比观测点与重投影点。

```python
import argparse
from pathlib import Path

import matplotlib
matplotlib.use("Agg")
import matplotlib.pyplot as plt
import numpy as np

from vslam.common import project, transform, write_json, write_ply
from vslam.geometry.two_view import estimate_pair


def main():
    parser = argparse.ArgumentParser()
    parser.add_argument("--input", required=True)
    parser.add_argument("--out", required=True)
    args = parser.parse_args()

    out = Path(args.out)
    out.mkdir(parents=True, exist_ok=True)

    # np.load 读取阶段一保存的匹配点和相机参数。
    with np.load(args.input) as data:
        result, report = estimate_pair(
            data["points0"], data["points1"], data["K"], data["distortion"]
        )
        for key in ["timestamp0", "timestamp1", "index0", "index1"]:
            result[key] = data[key].copy()

    np.savez_compressed(out / "geometry.npz", **result)
    write_ply(out / "points_c0.ply", result["points_c0"])
    write_json(out / "report.json", report)

    # 把三维点变换到第二帧，再投影回图像，检查重投影是否贴合观测点。
    points_c1 = transform(result["T_c1_c0"], result["points_c0"])
    projected = project(result["K"], points_c1)
    observed = result["uv1"]

    fig, ax = plt.subplots()
    ax.scatter(observed[:, 0], observed[:, 1], s=4, label="observed")
    ax.scatter(projected[:, 0], projected[:, 1], s=4, label="reprojected")
    ax.invert_yaxis()
    ax.set_aspect("equal")
    ax.legend()
    fig.savefig(out / "reprojection.png", dpi=150)
    plt.close(fig)


if __name__ == "__main__":
    main()
```

### 2.5 创建 `tests/test_geometry.py`

这不是检查实现有没有调用自身，而是用已知相机、点和反投影验证几何约定。

```python
import numpy as np
from scipy.spatial.transform import Rotation

from vslam.common import make_T, project, transform, backproject
from vslam.geometry.two_view import estimate_pair


def test_projection_roundtrip():
    # 投影后再按原深度反投影，应回到原始三维点。
    K = np.array([[500., 0, 320], [0, 500, 240], [0, 0, 1]])
    X = np.array([[0.2, -0.3, 2.], [-0.4, 0.5, 3.]])
    assert np.allclose(backproject(project(K, X), X[:, 2], K), X)

    # T 和 T 的逆变换连续作用后，应恢复原始点坐标。
    T = make_T(Rotation.from_rotvec([0.1, -0.05, 0.02]).as_matrix(), [0.4, 0.2, 0.1])
    assert np.allclose(transform(np.linalg.inv(T), transform(T, X)), X)


def test_two_view_known_scene():
    # 用已知三维点和相机运动生成两帧，验证恢复出的旋转和平移方向。
    rng = np.random.default_rng(2)
    X = rng.uniform([-1., -0.8, 3.], [1., 0.8, 6.], size=(150, 3))
    K = np.array([[500., 0, 320], [0, 500, 240], [0, 0, 1]])
    R = Rotation.from_rotvec([0.01, 0.04, -0.02]).as_matrix()
    t = np.array([-0.3, 0.01, 0.02])

    p0 = project(K, X)
    p1 = project(K, transform(make_T(R, t), X))
    result, report = estimate_pair(p0, p1, K, np.zeros(5))
    estimated = result["T_c1_c0"]

    angle = Rotation.from_matrix(estimated[:3, :3] @ R.T).magnitude()
    assert angle < 0.01
    assert np.dot(estimated[:3, 3], t / np.linalg.norm(t)) > 0.99
    assert report["median_reprojection_px"] < 0.1
```

### 3.1 创建 `src/vslam/data/association.py`

作用：把 RGB 与深度时间戳关联起来，并保证一张深度图不会被重复分配。`bisect_left()` / `bisect_right()` 在已排序的深度时间戳中快速定位容差区间；随后按时间差从小到大贪心选择对应。

```python
import argparse
from bisect import bisect_left, bisect_right
from pathlib import Path

from vslam.data.tum import read_index


def associate(rgb_rows, depth_rows, max_difference=0.02):
    # 先为每个 RGB 时间戳找出容差范围内的深度候选。
    depth_times = [row[0] for row in depth_rows]
    candidates = []

    for rgb_id, (rgb_time, _) in enumerate(rgb_rows):
        # bisect_left/right 在已排序时间戳中快速找到候选区间。
        left = bisect_left(depth_times, rgb_time - max_difference)
        right = bisect_right(depth_times, rgb_time + max_difference)

        for depth_id in range(left, right):
            gap = abs(rgb_time - depth_times[depth_id])
            candidates.append((gap, rgb_id, depth_id))

    # 按时间差从小到大匹配，并保证每张 RGB / 深度图只使用一次。
    candidates.sort()
    used_rgb, used_depth, selected = set(), set(), []

    for gap, rgb_id, depth_id in candidates:
        if rgb_id in used_rgb or depth_id in used_depth:
            continue

        selected.append((*rgb_rows[rgb_id], *depth_rows[depth_id]))
        used_rgb.add(rgb_id)
        used_depth.add(depth_id)

    # 输出顺序按 RGB 时间戳排列。
    return sorted(selected, key=lambda row: row[0])


def read_associations(path):
    # 读取关联文件，每行格式为：rgb_time rgb_path depth_time depth_path。
    rows = []
    for line in Path(path).read_text().splitlines():
        if not line.strip() or line.lstrip().startswith("#"):
            continue
        rgb_time, rgb_path, depth_time, depth_path = line.split()
        rows.append((float(rgb_time), rgb_path, float(depth_time), depth_path))
    return rows


def main():
    parser = argparse.ArgumentParser()
    parser.add_argument("--root", required=True)
    parser.add_argument("--out", required=True)
    parser.add_argument("--max-difference", type=float, default=0.02)
    args = parser.parse_args()

    root = Path(args.root)
    rgb_rows = read_index(root / "rgb.txt")
    depth_rows = read_index(root / "depth.txt")
    rows = associate(rgb_rows, depth_rows, args.max_difference)

    out = Path(args.out)
    out.parent.mkdir(parents=True, exist_ok=True)
    with out.open("w", encoding="utf-8") as f:
        for rgb_time, rgb_path, depth_time, depth_path in rows:
            f.write(
                f"{rgb_time:.6f} {rgb_path} "
                f"{depth_time:.6f} {depth_path}\n"
            )

    print(f"已生成 {len(rows)} 对 RGB-D 关联")
```

### 3.3 创建 `src/vslam/cli/run_vo.py`

作用：用上一帧深度建立 3D 点，再通过 ORB 匹配和 PnP 估计相邻帧位姿，最后累积成 TUM 轨迹。关键 API：`solvePnPRansac()` 从 3D-2D 对应中估计位姿并剔除外点；`solvePnPRefineLM()` 进一步细化位姿；`projectPoints()` 用于计算重投影误差。跟踪失败时停止，不补造后续位姿。

```python
import argparse
import csv
from pathlib import Path

import cv2
import numpy as np

from vslam.common import backproject, make_K, make_T, write_json, write_tum
from vslam.config import load_config
from vslam.data.association import read_associations
from vslam.vo.features import extract_orb, match_ratio, matched_points


def load_rgbd(root, row):
    # row 中第 2、4 项分别是 RGB 和深度图的相对路径。
    rgb = cv2.imread(str(root / row[1]), cv2.IMREAD_COLOR)
    depth = cv2.imread(str(root / row[3]), cv2.IMREAD_UNCHANGED)
    if rgb is None or depth is None:
        raise FileNotFoundError(f"读取图像失败：{row}")
    return rgb, depth


def main():
    parser = argparse.ArgumentParser()
    parser.add_argument("--config", required=True)
    parser.add_argument("--associations", required=True)
    parser.add_argument("--max-frames", type=int, default=100)
    parser.add_argument("--out", required=True)
    args = parser.parse_args()

    cfg = load_config(args.config)
    root, out = Path(cfg["dataset"]["root"]), Path(args.out)
    out.mkdir(parents=True, exist_ok=True)
    rows = read_associations(args.associations)
    if args.max_frames > 0:
        rows = rows[:args.max_frames]
    if len(rows) < 2:
        raise ValueError("至少需要两组 RGB-D 数据")

    K, vo = make_K(cfg["camera"]), cfg["vo"]
    if np.any(cfg["camera"]["distortion"]):
        raise ValueError("当前 RGB-D 基线使用零畸变模型")

    # 第一帧作为世界坐标系原点，初始相机位姿为单位矩阵。
    image0, depth0 = load_rgbd(root, rows[0])
    kp0, des0 = extract_orb(image0, cfg["orb"]["nfeatures"])
    poses, stamps, logs = [np.eye(4)], [rows[0][0]], []

    for row in rows[1:]:
        log = dict(timestamp=row[0], matches=0, inliers=0,
                   median_error_px=None, status="ok")
        try:
            image1, depth1 = load_rgbd(root, row)
            kp1, des1 = extract_orb(image1, cfg["orb"]["nfeatures"])

            # ORB 匹配得到两帧中的像素对应关系。
            matches = match_ratio(des0, des1, cfg["matching"]["ratio"])
            uv0, uv1 = matched_points(kp0, kp1, matches)
            log["matches"] = len(matches)

            # 从上一帧深度图读取深度，并反投影成三维点。
            pixels = np.rint(uv0).astype(int)
            z = depth0[pixels[:, 1], pixels[:, 0]].astype(float)
            z /= cfg["camera"]["depth_scale"]
            valid = (z >= vo["min_depth"]) & (z <= vo["max_depth"])
            points3d = backproject(uv0[valid], z[valid], K)
            points2d = uv1[valid]

            if len(points3d) < vo["min_correspondences"]:
                raise RuntimeError("有效深度对应点不足")

            # solvePnPRansac 用 3D-2D 对应关系估计当前帧相对位姿。
            ok, rvec, tvec, ids = cv2.solvePnPRansac(
                points3d, points2d, K, None,
                iterationsCount=200,
                reprojectionError=vo["reprojection_px"],
                confidence=0.999,
                flags=cv2.SOLVEPNP_EPNP
            )
            if not ok or ids is None:
                raise RuntimeError("PnP 求解失败")

            ids = ids.ravel()
            log["inliers"] = len(ids)
            if (len(ids) < vo["min_inliers"]
                    or len(ids) / len(points3d) < vo["min_inlier_ratio"]):
                raise RuntimeError("PnP 内点不足")

            # solvePnPRefineLM 在 RANSAC 内点上细化位姿。
            rvec, tvec = cv2.solvePnPRefineLM(
                points3d[ids], points2d[ids], K, None, rvec, tvec
            )
            projected, _ = cv2.projectPoints(points3d[ids], rvec, tvec, K, None)
            error = np.linalg.norm(projected.reshape(-1, 2) - points2d[ids], axis=1)
            log["median_error_px"] = float(np.median(error))

            if log["median_error_px"] > vo["reprojection_px"]:
                raise RuntimeError("重投影误差过大")

            # PnP 输出 T_c1_c0；轨迹是 T_wc，所以累乘相对位姿的逆矩阵。
            T_c1_c0 = make_T(cv2.Rodrigues(rvec)[0], tvec)
            poses.append(poses[-1] @ np.linalg.inv(T_c1_c0))
            stamps.append(row[0])
            kp0, des0, depth0 = kp1, des1, depth1

        except (RuntimeError, cv2.error) as error:
            log["status"] = f"lost: {error}"
            logs.append(log)
            break

        logs.append(log)

    # 保存每一步的匹配数、PnP 内点和重投影误差。
    with (out / "tracking.csv").open("w", newline="") as f:
        writer = csv.DictWriter(f, fieldnames=list(logs[0]))
        writer.writeheader()
        writer.writerows(logs)

    write_tum(out / "trajectory_tum.txt", stamps, poses)
    np.savez_compressed(out / "trajectory.npz", timestamps=stamps, poses_wc=poses)
    report = {
        "requested_frames": len(rows),
        "successful_poses": len(poses),
        "coverage": len(poses) / len(rows),
        "stopped_on_failure": logs[-1]["status"] != "ok"
    }
    write_json(out / "report.json", report)
    print(report)


if __name__ == "__main__":
    main()

```

## 5. 阶段 4：Bundle Adjustment

目标：先在独立合成场景中验证优化，再读取阶段 2A 的几何输出。第一版使用 SciPy 稀疏有限差分和非线性最小二乘，后续再推导解析雅可比或移植 Ceres。

### 5.1 创建 `src/vslam/optimization/ba.py`

作用：把相机位姿、三维点和二维观测放进同一个优化问题，最小化重投影残差。主要库：SciPy `least_squares()` 执行非线性最小二乘；`lil_matrix` 描述稀疏雅可比结构；`Rotation.from_rotvec()` 将旋转向量转换成旋转矩阵；OpenCV `Rodrigues()` 在旋转矩阵和旋转向量之间转换。输入观测必须是无畸变像素。

为消除 gauge freedom，固定第一相机位姿和第一个点在该坐标系中的 z。合成实验使用已知 z 锚定米制尺度；真实两帧使用初始 z 锚定已有任意尺度，不能因此声称恢复了米制尺度。

```python
import argparse
from pathlib import Path

import cv2
import numpy as np
from scipy.optimize import least_squares
from scipy.sparse import lil_matrix
from scipy.spatial.transform import Rotation

from vslam.common import project, write_json, write_ply


def synthetic(seed=0, outlier_fraction=0.0):
    # 构造 4 个相机视角和已知三维点，用于验证 BA。
    rng = np.random.default_rng(seed)
    K = np.array([[500., 0, 320], [0, 500, 240], [0, 0, 1]])
    X = rng.uniform([-1., -0.8, 3.], [1., 0.8, 5.], size=(60, 3))
    cameras = np.zeros((4, 6))  # 每台相机的旋转向量和位移向量
    cameras[:, 3] = -np.arange(4) * 0.25
    ci = np.repeat(np.arange(4), len(X))
    pi = np.tile(np.arange(len(X)), 4)

    clean = np.concatenate([project(K, X + c[3:]) for c in cameras])
    uv = clean + rng.normal(0, 0.5, clean.shape)
    bad = rng.choice(len(uv), int(len(uv) * outlier_fraction), replace=False)
    uv[bad] += rng.normal(0, 25, (len(bad), 2))

    # 给初始位姿和三维点加小扰动，模拟真实优化的初值误差。
    cameras[1:, :3] += rng.normal(0, 0.01, (3, 3))
    cameras[1:, 3:] += rng.normal(0, 0.025, (3, 3))
    initial_X = X + rng.normal(0, 0.03, X.shape)
    initial_X[0, 2] = X[0, 2]  # 固定一个深度，消除尺度自由度。
    return K, cameras, initial_X, ci, pi, uv, clean


def from_pair(path):
    # 将两帧几何结果转成两相机 BA 的输入格式。
    with np.load(path) as d:
        X = d["points_c0"].copy()
        ids = np.linspace(0, len(X) - 1, min(120, len(X)), dtype=int)
        X = X[ids]
        T = d["T_c1_c0"]

        cameras = np.zeros((2, 6))
        cameras[1, :3] = cv2.Rodrigues(T[:3, :3])[0].ravel()
        cameras[1, 3:] = T[:3, 3]
        uv = np.concatenate([d["uv0"][ids], d["uv1"][ids]])

        return (d["K"].copy(), cameras, X, np.repeat([0, 1], len(X)),
                np.tile(np.arange(len(X)), 2), uv, None)


def optimize(K, cameras0, points0, ci, pi, uv, loss):
    # 固定第一台相机，并固定第一个点的 z，消除 BA 的 gauge freedom。
    ncam = len(cameras0)
    camera_variables = 6 * (ncam - 1)
    free = np.delete(np.arange(points0.size), 2)
    x0 = np.r_[cameras0[1:].ravel(), points0.ravel()[free]]

    def unpack(x):
        cameras = cameras0.copy()
        cameras[1:] = x[:camera_variables].reshape(-1, 6)
        points = points0.ravel().copy()
        points[free] = x[camera_variables:]
        return cameras, points.reshape(-1, 3)

    def predict(x):
        cameras, points = unpack(x)
        R = Rotation.from_rotvec(cameras[:, :3]).as_matrix()
        Xc = np.einsum("nij,nj->ni", R[ci], points[pi]) + cameras[ci, 3:]
        return project(K, Xc), Xc[:, 2]

    def residual(x):
        # 每条观测提供 u、v 两个像素残差。
        return (predict(x)[0] - uv).ravel()

    # 稀疏雅可比：一条观测只依赖对应相机的 6 个参数和点的 3 个坐标。
    sparsity = lil_matrix((2 * len(uv), len(x0)), dtype=int)
    point_col = {old: camera_variables + j for j, old in enumerate(free)}
    for obs, (c, pt) in enumerate(zip(ci, pi)):
        rows = slice(2 * obs, 2 * obs + 2)
        if c > 0:
            sparsity[rows, 6 * (c - 1):6 * c] = 1
        for old in range(3 * pt, 3 * pt + 3):
            if old in point_col:
                sparsity[rows, point_col[old]] = 1

    # least_squares 求非线性最小二乘；Huber 损失可降低外点影响。
    result = least_squares(
        residual, x0, jac_sparsity=sparsity.tocsr(),
        method="trf", x_scale="jac", loss=loss, f_scale=1.0
    )
    pred0 = predict(x0)[0]
    pred1 = predict(result.x)[0]
    cameras, points = unpack(result.x)

    report = {
        "success": bool(result.success),
        "message": str(result.message),
        "evaluations": int(result.nfev),
        "loss": loss,
        "before_rmse_px": float(np.sqrt(np.mean((pred0 - uv) ** 2))),
        "after_rmse_px": float(np.sqrt(np.mean((pred1 - uv) ** 2))),
        "before_median_px": float(np.median(np.linalg.norm(pred0 - uv, axis=1))),
        "after_median_px": float(np.median(np.linalg.norm(pred1 - uv, axis=1)))
    }
    return cameras, points, pred0, pred1, report


def main():
    parser = argparse.ArgumentParser()
    parser.add_argument("--pair", help="输入 geometry.npz；不提供则使用合成数据")
    parser.add_argument("--loss", choices=["linear", "huber"], default="huber")
    parser.add_argument("--outliers", type=float, default=0.0)
    parser.add_argument("--out", required=True)
    args = parser.parse_args()

    data = from_pair(args.pair) if args.pair else synthetic(outlier_fraction=args.outliers)
    K, cams, points, ci, pi, uv, clean = data
    c1, X1, pred0, pred1, report = optimize(K, cams, points, ci, pi, uv, args.loss)
    report["scale"] = "arbitrary" if args.pair else "synthetic metric anchor"

    if clean is not None:
        report["clean_rmse_before"] = float(np.sqrt(np.mean((pred0 - clean) ** 2)))
        report["clean_rmse_after"] = float(np.sqrt(np.mean((pred1 - clean) ** 2)))

    out = Path(args.out)
    out.mkdir(parents=True, exist_ok=True)
    np.savez_compressed(
        out / "ba.npz", K=K, cameras_before=cams, cameras_after=c1,
        points_before=points, points_after=X1,
        camera_indices=ci, point_indices=pi, observations=uv
    )
    write_ply(out / "before.ply", points)
    write_ply(out / "after.ply", X1)
    write_json(out / "report.json", report)
    print(report)


if __name__ == "__main__":
    main()
```

### 6.2 创建 `src/vslam/reconstruction/select_frames.py`

作用：按帧间隔抽取图片，并保存新文件名、原始路径和时间戳的对应表。OpenCV `Laplacian()` 的方差用于粗略衡量清晰度；`shutil.copy2()` 复制图片并保留文件时间信息。清晰度筛选只是粗筛，仍需人工检查模糊和视角覆盖。

```python
import argparse
import csv
import shutil
from pathlib import Path

import cv2
from vslam.data.tum import read_index


def main():
    parser = argparse.ArgumentParser()
    parser.add_argument("--root", required=True)
    parser.add_argument("--out", required=True)
    parser.add_argument("--stride", type=int, default=8)
    parser.add_argument("--max-frames", type=int, default=100)
    parser.add_argument("--min-sharpness", type=float, default=0.0)
    args = parser.parse_args()

    root, out = Path(args.root), Path(args.out)
    image_dir = out / "input"
    image_dir.mkdir(parents=True, exist_ok=True)
    if any(image_dir.iterdir()):
        raise FileExistsError("input 目录已有图片，请换一个场景输出目录")
    records = []

    for i, (ts, rel) in enumerate(read_index(root / "rgb.txt")):
        if i % args.stride:
            continue

        image = cv2.imread(str(root / rel), cv2.IMREAD_GRAYSCALE)
        if image is None:
            raise FileNotFoundError(root / rel)
        # Laplacian 方差用于粗略衡量清晰度，值越低通常越模糊。
        sharpness = cv2.Laplacian(image, cv2.CV_64F).var()
        if sharpness < args.min_sharpness:
            continue

        name = f"{i:06d}{Path(rel).suffix}"
        shutil.copy2(root / rel, image_dir / name)
        records.append(dict(name=name, timestamp=ts, original=rel, sharpness=sharpness))
        if len(records) >= args.max_frames:
            break

    if not records:
        raise RuntimeError("没有选中图像，请调整 stride 或清晰度阈值")

    # CSV 保留新文件名与原始帧的映射关系。
    with (out / "frames.csv").open("w", newline="") as f:
        writer = csv.DictWriter(f, fieldnames=list(records[0]))
        writer.writeheader()
        writer.writerows(records)
    print(f"已选择 {len(records)} 张图像")


if __name__ == "__main__":
    main()


```

### 6.3 创建 `scripts/run_colmap.sh`

作用：分为 `sfm` 和 `undistort` 两步：前者提取 SIFT 特征、进行穷举匹配并运行 mapper；后者把选定模型去畸变并整理成 3DGS 所需的 `sparse/0` 目录。示例使用 COLMAP 常见的 `SiftExtraction.use_gpu`、`SiftMatching.use_gpu` 参数关闭 GPU，先用 CPU 跑通流程。若本机版本的 `colmap ... -h` 显示参数前缀不同，按帮助信息替换这两个参数名即可。

```bash
#!/usr/bin/env bash
set -euo pipefail

SCENE=$(realpath "${1:?用法: bash scripts/run_colmap.sh 场景目录 sfm或undistort [模型编号]}")
MODE=${2:-sfm}
MODEL=${3:-0}

# sfm：提取特征、匹配并重建稀疏模型。
if [[ "$MODE" == "sfm" ]]; then
  mkdir -p "$SCENE/distorted/sparse"

  # 这里关闭 SIFT GPU，先用 CPU 跑通流程。
  colmap feature_extractor \
    --database_path "$SCENE/distorted/database.db" \
    --image_path "$SCENE/input" \
    --ImageReader.single_camera 1 \
    --ImageReader.camera_model OPENCV \
    --SiftExtraction.use_gpu 0

  colmap exhaustive_matcher \
    --database_path "$SCENE/distorted/database.db" \
    --SiftMatching.use_gpu 0

  colmap mapper \
    --database_path "$SCENE/distorted/database.db" \
    --image_path "$SCENE/input" \
    --output_path "$SCENE/distorted/sparse"

  echo "SfM 完成。先检查 sparse 下的模型，再执行 undistort。"

# undistort：选择一个稀疏模型，生成去畸变图片和 3DGS 所需目录。
elif [[ "$MODE" == "undistort" ]]; then
  colmap model_analyzer --path "$SCENE/distorted/sparse/$MODEL"

  colmap image_undistorter \
    --image_path "$SCENE/input" \
    --input_path "$SCENE/distorted/sparse/$MODEL" \
    --output_path "$SCENE" \
    --output_type COLMAP

  # 3DGS 默认从 sparse/0 读取相机、图像和稀疏点文件。
  mkdir -p "$SCENE/sparse/0"
  for name in cameras images points3D; do
    cp "$SCENE/sparse/$name.bin" "$SCENE/sparse/0/$name.bin"
  done
else
  echo "MODE 必须是 sfm 或 undistort"
  exit 1
fi
```

### 7.2 创建 `src/vslam/semantic/build_map.py`

作用：用 SAM 生成区域 mask，用 CLIP 编码区域语义，再结合深度和相机位姿把采样点投到世界坐标，最后按体素融合。SAM `SamAutomaticMaskGenerator.generate()` 生成候选区域；CLIP `preprocess()` 处理区域裁剪，`encode_image()` 提取图像特征；PyTorch `inference_mode()` 关闭梯度计算，减少推理开销。输出最终语义点云、特征数组和融合报告。

```python
    if np.any(cfg["camera"]["distortion"]):

    )
    model, preprocess = clip.load(
        "ViT-B/32", device="cpu", download_root=args.clip_cache
    )
    model.eval()

    # 每个体素累计位置、颜色、CLIP 特征、总权重和观测次数。
    voxels = {}
    skipped_pose, frame_reports = 0, []

    for row in rows:
        T_wc = nearest_pose(row[0], times, poses, tolerance=0.02)
        if T_wc is None:
            skipped_pose += 1
            continue

        bgr, depth = load_rgbd(root, row)
        rgb = cv2.cvtColor(bgr, cv2.COLOR_BGR2RGB)

        # SAM 的 generate() 返回 mask、区域面积和 predicted_iou。
        sam.to(args.device)
        with torch.inference_mode():
            masks = mask_generator.generate(rgb)
        sam.cpu()

        # 去掉过小或几乎覆盖整张图的区域，小区域优先分配重叠像素。
        masks = sorted(
            [m for m in masks if 200 <= m["area"] <= 0.8 * rgb.shape[0] * rgb.shape[1]],
            key=lambda m: m["area"]
        )[:50]
        labels = np.full(depth.shape, -1, dtype=np.int32)
        features, weights = [], []

        model.to(args.device)
        for region_id, mask in enumerate(masks):
            seg = mask["segmentation"]
            yy, xx = np.where(seg)
            x0, x1, y0, y1 = xx.min(), xx.max() + 1, yy.min(), yy.max() + 1
            crop = rgb[y0:y1, x0:x1].copy()
            crop[~seg[y0:y1, x0:x1]] = 127

            # preprocess 把 PIL 图片转成 CLIP 所需的张量；encode_image 提取图像特征。
            image = preprocess(Image.fromarray(crop)).unsqueeze(0).to(args.device)
            with torch.inference_mode():
                feature = model.encode_image(image).float()
                feature /= feature.norm(dim=-1, keepdim=True)

            features.append(feature.cpu().numpy()[0])
            weights.append(max(0.05, float(mask["predicted_iou"])))
            labels[seg & (labels < 0)] = region_id

        model.cpu()
        if not features:
            continue

        features = np.asarray(features, dtype=np.float32)

        # 每隔 pixel_stride 个像素采样一次，降低反投影计算量。
        yy, xx = np.mgrid[
            0:cfg["camera"]["height"]:args.pixel_stride,
            0:cfg["camera"]["width"]:args.pixel_stride
        ]
        xx, yy = xx.ravel(), yy.ravel()
        z = depth[yy, xx].astype(float) / cfg["camera"]["depth_scale"]
        region_ids = labels[yy, xx]

        valid = (
            (z >= cfg["vo"]["min_depth"])
            & (z <= cfg["vo"]["max_depth"])
            & (region_ids >= 0)
        )
        xx, yy, z, region_ids = xx[valid], yy[valid], z[valid], region_ids[valid]

        # 先反投影到相机坐标，再用 T_wc 转到世界坐标。
        points = transform(T_wc, backproject(np.column_stack([xx, yy]), z, K))
        voxel_ids = np.floor(points / args.voxel_size).astype(int)

        # 同一帧同一体素只累计一次，避免重复像素提高该帧权重。
        seen = set()
        for voxel_id, point, color, rid in zip(voxel_ids, points, rgb[yy, xx], region_ids):
            key = tuple(voxel_id)
            if key in seen:
                continue
            seen.add(key)

            if key not in voxels:
                voxels[key] = [
                    np.zeros(3), np.zeros(3), np.zeros(features.shape[1]), 0.0, 0
                ]
            item = voxels[key]
            weight = weights[rid]
            item[0] += weight * point
            item[1] += weight * color
            item[2] += weight * features[rid]
            item[3] += weight
            item[4] += 1

        frame_reports.append({
            "timestamp": row[0],
            "regions": len(features),
            "sampled_points": len(points)
        })

    # 对每个体素的累计值求加权平均，得到语义点云。
    if not voxels:
        raise RuntimeError("没有有效语义点，请检查深度、姿态和 SAM mask")
    values = list(voxels.values())
    points = np.array([v[0] / v[3] for v in values])
    colors = np.array([v[1] / v[3] for v in values]).astype(np.uint8)
    features = np.array([v[2] / v[3] for v in values], dtype=np.float32)
    features /= np.linalg.norm(features, axis=1, keepdims=True)
    observations = np.array([v[4] for v in values], dtype=np.int32)

    np.savez_compressed(
        out / "semantic_map.npz",
        points=points, colors=colors, features=features,
        observations=observations, clip_model="ViT-B/32",
        trajectory=str(Path(args.trajectory).resolve()),
        voxel_size=args.voxel_size, units="meters"
    )
    write_ply(out / "semantic_map.ply", points, colors)
    write_json(out / "report.json", {
        "fused_frames": len(frame_reports),
        "skipped_missing_pose": skipped_pose,
        "voxels": len(points),
        "frame_reports": frame_reports
    })


if __name__ == "__main__":
    main()


```

这版实现的是“区域裁剪的 CLIP 特征传播到可见深度点”，不是像素级特征，也不是实例跟踪。没有 mask 的点不会被赋予语义；物体边界、遮挡和区域裁剪偏差都会带来误差。体素大小越小，几何细节越多，但内存占用也越高。

### 7.3 创建 `src/vslam/semantic/query.py`

作用：将文本查询和点云中的 CLIP 特征进行比较，保存相似度并导出高亮点云。`clip.tokenize()` 把文本转成模型输入，`encode_text()` 提取文本特征；归一化后的点积是余弦相似度，可用于排序和阈值筛选，但不是校准概率。

```python
import argparse
from pathlib import Path

import clip
import numpy as np
import torch

from vslam.common import write_json, write_ply


def main():
    parser = argparse.ArgumentParser()
    parser.add_argument("--map", required=True)
    parser.add_argument("--text", default="chair")
    parser.add_argument("--alternatives", nargs="+", default=["table", "wall", "floor", "monitor"])
    parser.add_argument("--threshold", type=float, default=0.20)
    parser.add_argument("--min-observations", type=int, default=1)
    parser.add_argument("--clip-cache", default="checkpoints/clip")
    parser.add_argument("--out", required=True)
    args = parser.parse_args()

    # 读取语义点云，以及每个点对应的 CLIP 图像特征。
    with np.load(args.map) as d:
        points, features = d["points"].copy(), d["features"].copy()
        counts = d["observations"].copy()
        model_name = str(d["clip_model"])

    model, _ = clip.load(model_name, device="cpu", download_root=args.clip_cache)

    # 把查询词和备选类别编码到 CLIP 的文本特征空间。
    names = [args.text] + [x for x in args.alternatives if x != args.text]
    tokens = clip.tokenize([f"a photo of a {x}" for x in names])
    with torch.inference_mode():
        text_features = model.encode_text(tokens).float()
        text_features /= text_features.norm(dim=-1, keepdim=True)

    # 归一化特征点积即余弦相似度；不是经过校准的概率。
    scores = features @ text_features.numpy().T
    selected = (
        (scores.argmax(axis=1) == 0)
        & (scores[:, 0] >= args.threshold)
        & (counts >= args.min_observations)
    )

    colors = np.full((len(points), 3), 100, dtype=np.uint8)
    colors[selected] = [255, 60, 30]

    out = Path(args.out)
    out.mkdir(parents=True, exist_ok=True)
    write_ply(out / "query.ply", points, colors)
    np.savez_compressed(
        out / "scores.npz", scores=scores,
        selected=selected, names=np.array(names)
    )
    write_json(out / "report.json", {
        "query": args.text,
        "selected_points": int(selected.sum()),
        "total_points": len(points),
        "threshold": args.threshold,
        "candidates": names
    })


if __name__ == "__main__":
    main()
```
